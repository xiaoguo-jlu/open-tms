  }).then(res =&gt; {

    const banks = new Map()
    const records = res.data?.records || []
    records.forEach(r =&gt; {
      if (r.bankId &amp;&amp; !banks.has(r.bankId)) {
        banks.set(r.bankId, { id: r.bankId, name: `银行#${r.bankId}` })

  })
}
